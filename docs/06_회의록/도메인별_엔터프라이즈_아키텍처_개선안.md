```java
// 비밀번호 정책
public class PasswordPolicy {
    private static final int MIN_LENGTH = 10;
    private static final Pattern COMPLEXITY =
        Pattern.compile("^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d)(?=.*[@$!%*?&])[A-Za-z\\d@$!%*?&]{10,}$");

    public void validate(String password, Member member) {
        // 최소 길이
        if (password.length() < MIN_LENGTH) {
            throw new InvalidPasswordException("최소 10자 이상");
        }

        // 복잡도
        if (!COMPLEXITY.matcher(password).matches()) {
            throw new InvalidPasswordException("대소문자, 숫자, 특수문자 포함 필수");
        }

        // 과거 비밀번호 재사용 금지 (최근 3개)
        List<String> history = passwordHistoryRepository
            .findRecentPasswords(member.getId(), 3);

        for (String oldPasswordHash : history) {
            if (passwordEncoder.matches(password, oldPasswordHash)) {
                throw new InvalidPasswordException("최근 사용한 비밀번호는 재사용 불가");
            }
        }
    }
}

// 이상 로그인 감지
public class AnomalyDetector {

    public boolean detectAnomaly(LoginAttempt attempt, Member member) {
        // 1. 새로운 국가에서 로그인
        String lastCountry = loginHistoryRepository
            .findLastLoginCountry(member.getId());

        if (!attempt.getCountry().equals(lastCountry)) {
            sendSecurityAlert(member, "새로운 위치에서 로그인 시도: " + attempt.getCountry());
            return true;
        }

        // 2. 불가능한 이동 (Impossible Travel)
        LoginHistory lastLogin = loginHistoryRepository
            .findLastLogin(member.getId());

        if (isImpossibleTravel(lastLogin, attempt)) {
            sendSecurityAlert(member, "물리적으로 불가능한 위치 이동 감지");
            return true;
        }

        // 3. 비정상적 시간대 (평소와 다른 시간)
        if (isUnusualTime(member.getId(), attempt.getTimestamp())) {
            // 추가 인증 요구 (MFA)
            return true;
        }

        return false;
    }
}
```

```java
// AS-IS (PostgreSQL LIKE)
@Query("SELECT p FROM Product p WHERE p.name LIKE %?1%")
List<Product> searchByName(String keyword);
// → 느림, 인덱스 미사용, 형태소 분석 불가

// TO-BE (Elasticsearch)
public class ProductSearchService {

    @Autowired
    private ElasticsearchOperations elasticsearchOperations;

    public SearchPage<Product> search(ProductSearchQuery query) {
        BoolQueryBuilder boolQuery = QueryBuilders.boolQuery();

        // 1. 키워드 검색 (형태소 분석)
        if (query.getKeyword() != null) {
            boolQuery.should(
                QueryBuilders.matchQuery("name", query.getKeyword())
                    .boost(3.0f) // 상품명 가중치 높게
            );
            boolQuery.should(
                QueryBuilders.matchQuery("description", query.getKeyword())
                    .boost(1.0f)
            );
        }

        // 2. 필터 (브랜드, 카테고리, 가격대)
        if (query.getBrand() != null) {
            boolQuery.filter(
                QueryBuilders.termQuery("brand", query.getBrand())
            );
        }

        if (query.getPriceRange() != null) {
            boolQuery.filter(
                QueryBuilders.rangeQuery("price")
                    .gte(query.getPriceRange().getMin())
                    .lte(query.getPriceRange().getMax())
            );
        }

        // 3. 정렬
        NativeSearchQuery searchQuery = new NativeSearchQueryBuilder()
            .withQuery(boolQuery)
            .withSort(SortBuilders.scoreSort().order(SortOrder.DESC)) // 관련도순
            .withSort(SortBuilders.fieldSort("createdAt").order(SortOrder.DESC)) // 최신순
            .withPageable(query.getPageable())
            .build();

        return elasticsearchOperations.searchForPage(searchQuery, Product.class);
    }
}
```

#### 2.2.3 재고 관리 및 동시성 제어 약함

**현재 문제:**
- 재고 차감 시 Race Condition 가능
- 분산 환경에서 동시 주문 시 재고 초과 판매 위험

**표준:** Redis Lua Script + Optimistic Locking

```java
// Redis Lua Script (원자적 재고 차감)
@Component
public class StockService {

    @Autowired
    private StringRedisTemplate redisTemplate;

    private static final String DECREASE_STOCK_SCRIPT =
        """
        local stock = redis.call('GET', KEYS[1])
        if not stock then
            return -1  -- 재고 정보 없음
        end

        stock = tonumber(stock)
        local quantity = tonumber(ARGV[1])

        if stock >= quantity then
            redis.call('DECRBY', KEYS[1], quantity)
            return stock - quantity  -- 남은 재고
        else
            return -2  -- 재고 부족
        end
        """;

    public boolean decreaseStock(UUID productId, int quantity) {
        String key = "product:stock:" + productId;

        Long result = redisTemplate.execute(
            RedisScript.of(DECREASE_STOCK_SCRIPT, Long.class),
            List.of(key),
            String.valueOf(quantity)
        );

        if (result == null || result == -2) {
            throw new OutOfStockException("재고 부족");
        }

        // 비동기로 DB 동기화
        syncStockToDatabase(productId, result);

        return true;
    }

    // DB와 Redis 재고 동기화 (이벤트 기반)
    @Async
    private void syncStockToDatabase(UUID productId, long newStock) {
        productRepository.updateStock(productId, newStock);
    }
}
```

```java
    public List<Product> findFrequentlyBought(UUID productId) {

            .range(cacheKey, 0, 5);

        if (cachedIds != null && !cachedIds.isEmpty()) {
            return productRepository.findAllById(cachedIds);
        }

        // 캐시 미스 시 계산
        return calculateAndCache(productId);
    }
```

```java
public class BidEvent {

    private UUID eventId;

    private String auctionId;
    private Long bidderId;
    private BigDecimal amount;
    private LocalDateTime timestamp;

    @Enumerated(EnumType.STRING)
    private BidEventType type; // PLACED, REJECTED, OUTBID

    private String reason; // 거절 사유
}

// Redis Stream으로 입찰 이벤트 처리
public class BidStreamProcessor {

    @Autowired
    private StringRedisTemplate redisTemplate;

    public void processBid(BidCommand command) {
        String streamKey = "auction:bids:" + command.getAuctionId();

        // 1. Redis Stream에 이벤트 추가 (초고속)
        Map<String, String> event = Map.of(
            "bidderId", command.getBidderId().toString(),
            "amount", command.getAmount().toString(),
            "timestamp", Instant.now().toString()
        );

        RecordId recordId = redisTemplate.opsForStream()
            .add(streamKey, event);

        // 2. 입찰 검증 (비동기)
        CompletableFuture.runAsync(() -> validateAndProcess(command));

        // 3. 즉시 응답 (낙관적 업데이트)
        return BidResult.pending(recordId.getValue());
    }

    // Consumer Group으로 병렬 처리
    @StreamListener(value = "auction:bids:*", group = "bid-processors")
    public void handleBidEvent(MapRecord<String, String, String> record) {
        String auctionId = extractAuctionId(record.getStream());
        BidEvent event = parseBidEvent(record.getValue());

        // 낙관적 잠금으로 검증
        Auction auction = auctionRepository.findByIdForUpdate(auctionId);

        if (event.getAmount().compareTo(auction.getCurrentBid().add(auction.getMinIncrement())) >= 0) {
            // 유효한 입찰
            auction.updateCurrentBid(event.getAmount(), event.getBidderId());
            auctionRepository.save(auction);

            // WebSocket으로 실시간 브로드캐스트
            webSocketService.broadcast(auctionId, BidUpdate.success(event));
        } else {
            // 무효한 입찰
            webSocketService.sendToUser(event.getBidderId(), BidUpdate.rejected("금액 부족"));
        }
    }
}
```

#### 3.2.2 자동 입찰 (Proxy Bidding) 부재

**표준:** eBay의 자동 입찰 시스템

사용자가 "최대 100만원까지 자동 입찰"을 설정하면, 시스템이 대신 입찰해줌.

```java
@Entity
public class ProxyBid {
    @Id
    private UUID id;

    private String auctionId;
    private Long bidderId;
    private BigDecimal maxAmount;  // 최대 입찰 금액
    private BigDecimal currentBid; // 현재까지 집행된 금액
    private boolean active;
}

public class ProxyBidService {

    // 타인이 입찰하면 자동으로 대응
    @EventListener
    public void onBidPlaced(BidPlacedEvent event) {
        List<ProxyBid> proxyBids = proxyBidRepository
            .findActiveByAuction(event.getAuctionId());

        for (ProxyBid proxy : proxyBids) {
            if (proxy.getBidderId().equals(event.getBidderId())) {
                continue; // 본인 입찰은 스킵
            }

            // 타인 입찰보다 minIncrement만큼 높게 자동 입찰
            BigDecimal counterBid = event.getAmount()
                .add(auction.getMinIncrement());

            if (counterBid.compareTo(proxy.getMaxAmount()) <= 0) {
                bidService.placeBid(BidCommand.builder()
                    .auctionId(event.getAuctionId())
                    .bidderId(proxy.getBidderId())
                    .amount(counterBid)
                    .automatic(true)
                    .build());
            } else {
                // 최대 금액 도달
                proxy.setActive(false);
                notifyMaxAmountReached(proxy);
            }
        }
    }
}
```

```java
public class AntiSniperService {

    private static final Duration EXTENSION_WINDOW = Duration.ofMinutes(5);
    private static final Duration EXTENSION_DURATION = Duration.ofMinutes(5);

    @EventListener
    public void onBidPlaced(BidPlacedEvent event) {
        Auction auction = auctionRepository.findById(event.getAuctionId());

        Duration timeUntilEnd = Duration.between(
            LocalDateTime.now(),
            auction.getEndsAt()

            auction.setEndsAt(auction.getEndsAt().plus(EXTENSION_DURATION));
            auctionRepository.save(auction);

            // 참여자들에게 알림
            webSocketService.broadcast(
                event.getAuctionId(),
                AuctionExtendedEvent.of(auction.getEndsAt())

        }
```

```java
@Configuration
public class AuctionExpirationConfig {

    @Bean
    public RedisMessageListenerContainer redisContainer() {

            new RedisMessageListenerContainer();

            if (expiredKey.startsWith("auction:expiration:")) {
                String auctionId = expiredKey.replace("auction:expiration:", "");
                auctionCloseService.closeAuction(auctionId);
            }

// 경매 시작 시 만료 키 설정
public class AuctionLifecycleService {

    public void scheduleAuctionExpiration(Auction auction) {
        String key = "auction:expiration:" + auction.getId();

        Duration ttl = Duration.between(
            LocalDateTime.now(),
            auction.getEndsAt()
        );

        // TTL 설정 (자동 만료 시 이벤트 발생)
        redisTemplate.opsForValue().set(
            key,
            auction.getId(),
            ttl
        );
    }
}

// 종료 처리 멱등성 보장
public class AuctionCloseService {

    @Transactional
    public void closeAuction(String auctionId) {
        Auction auction = auctionRepository
            .findByIdWithLock(auctionId);

        // 멱등성: 이미 종료된 경매는 스킵
        if (auction.getStatus() == AuctionStatus.ENDED) {
            log.info("Auction {} already closed", auctionId);
            return;
        }

        // 낙찰자 결정
        Optional<Bid> winningBid = bidRepository
            .findTopByAuctionIdOrderByAmountDesc(auctionId);

        if (winningBid.isPresent()) {
            auction.close(winningBid.get().getBidderId());

            // 낙찰 이벤트 발행 (Order 생성)
            eventPublisher.publish(AuctionWonEvent.of(auction, winningBid.get()));
        } else {
            auction.closeWithoutWinner();
        }

        auctionRepository.save(auction);
    }
}
```

```java
@Component
public class OrderSagaOrchestrator {

    public OrderResult createOrder(CreateOrderCommand command) {
        SagaInstance saga = sagaInstanceRepository.save(
            new SagaInstance(UUID.randomUUID(), command)
        );

        try {
            // Step 1: 재고 차감
            StockResult stockResult = executeStep(saga,
                () -> productService.decreaseStock(command.getProductId(), command.getQuantity()),
                () -> productService.increaseStock(command.getProductId(), command.getQuantity()) // 보상
            );

            // Step 2: 주문 생성
            Order order = executeStep(saga,
                () -> orderRepository.save(Order.create(command)),
                () -> orderRepository.delete(order.getId()) // 보상
            );

            // Step 3: 결제 처리
            PaymentResult paymentResult = executeStep(saga,
                () -> paymentService.charge(order.getTotalAmount(), command.getMemberId()),
                () -> paymentService.refund(paymentResult.getPaymentId()) // 보상

    }
```

#### 4.2.2 주문 상태 관리 State Machine 부재

**현재 문제:**
- 주문 상태 전이 규칙이 명확하지 않음
- 잘못된 상태 전이 가능 (예: SHIPPED → PENDING)

**표준:** Spring State Machine

```java
// 주문 상태 정의
public enum OrderState {
    PENDING,       // 주문 생성
    PAID,          // 결제 완료
    CONFIRMED,     // 판매자 확인
    PREPARING,     // 상품 준비 중
    SHIPPED,       // 배송 시작
    DELIVERED,     // 배송 완료
    CANCELLED,     // 취소
    REFUNDED       // 환불 완료
}

public enum OrderEvent {
    PAY,
    CONFIRM,
    PREPARE,
    SHIP,
    DELIVER,
    CANCEL,
    REFUND
}

// State Machine 설정
@Configuration
@EnableStateMachine
public class OrderStateMachineConfig extends StateMachineConfigurerAdapter<OrderState, OrderEvent> {

    @Override
    public void configure(StateMachineStateConfigurer<OrderState, OrderEvent> states) throws Exception {
        states
            .withStates()
            .initial(OrderState.PENDING)
            .states(EnumSet.allOf(OrderState.class));
    }

    @Override
    public void configure(StateMachineTransitionConfigurer<OrderState, OrderEvent> transitions) throws Exception {
        transitions
            // PENDING → PAID (결제 완료)
            .withExternal()
                .source(OrderState.PENDING).target(OrderState.PAID)
                .event(OrderEvent.PAY)
                .guard(paymentGuard())
                .action(paymentAction())
            .and()
            // PAID → CONFIRMED (판매자 확인)
            .withExternal()
                .source(OrderState.PAID).target(OrderState.CONFIRMED)
                .event(OrderEvent.CONFIRM)
            .and()
            // CONFIRMED → PREPARING
            .withExternal()
                .source(OrderState.CONFIRMED).target(OrderState.PREPARING)
                .event(OrderEvent.PREPARE)
            .and()
            // PREPARING → SHIPPED
            .withExternal()
                .source(OrderState.PREPARING).target(OrderState.SHIPPED)
                .event(OrderEvent.SHIP)
                .action(shippingAction())
            .and()
            // SHIPPED → DELIVERED
            .withExternal()
                .source(OrderState.SHIPPED).target(OrderState.DELIVERED)
                .event(OrderEvent.DELIVER)
            .and()
            // 취소 (여러 상태에서 가능)
            .withExternal()
                .source(OrderState.PENDING).target(OrderState.CANCELLED)
                .event(OrderEvent.CANCEL)
            .and()
            .withExternal()
                .source(OrderState.PAID).target(OrderState.CANCELLED)
                .event(OrderEvent.CANCEL)
                .action(refundAction());
    }

    @Bean
    public Guard<OrderState, OrderEvent> paymentGuard() {
        return context -> {
            Order order = context.getExtendedState().get("order", Order.class);
            return order.getTotalAmount().compareTo(BigDecimal.ZERO) > 0;
        };
    }

    @Bean
    public Action<OrderState, OrderEvent> paymentAction() {
        return context -> {
            Order order = context.getExtendedState().get("order", Order.class);
            paymentService.charge(order);
            log.info("Payment processed for order {}", order.getId());
        };
    }
}
```

```java
public class IdempotentRequest {
    @Id
    private String idempotencyKey; // 클라이언트가 생성한 UUID

    private UUID memberId;
    private String endpoint; // "/api/payments"

    @Convert(converter = JsonConverter.class)
    private Map<String, Object> requestBody;

    @Convert(converter = JsonConverter.class)
    private Map<String, Object> response;

    private int statusCode;
    private LocalDateTime createdAt;
    private LocalDateTime expiresAt; // 24시간 후
}

@Aspect
@Component
public class IdempotencyAspect {

    @Around("@annotation(idempotent)")
    public Object handleIdempotency(ProceedingJoinPoint pjp, Idempotent idempotent) throws Throwable {
        HttpServletRequest request =
            ((ServletRequestAttributes) RequestContextHolder.getRequestAttributes()).getRequest();

        String idempotencyKey = request.getHeader("Idempotency-Key");

        if (idempotencyKey == null) {
            throw new MissingIdempotencyKeyException();
        }

        // 이미 처리된 요청인지 확인
        Optional<IdempotentRequest> existing =
            idempotentRequestRepository.findById(idempotencyKey);

        if (existing.isPresent()) {
            // 동일 요청의 응답 반환
            log.info("Returning cached response for idempotency key: {}", idempotencyKey);
            return existing.get().getResponse();
        }

        // 신규 요청 처리
        Object result = pjp.proceed();

        // 결과 저장 (24시간)
        idempotentRequestRepository.save(
            IdempotentRequest.builder()
                .idempotencyKey(idempotencyKey)
                .response(result)
                .expiresAt(LocalDateTime.now().plusHours(24))
                .build()
        );

        return result;
    }
}

// 사용
@PostMapping("/payments")
@Idempotent
public PaymentResponse charge(@RequestBody ChargeRequest request) {
    return paymentService.charge(request);
}
```

```java
@Scheduled(cron = "0 0 2 1 * ?") // 매월 1일 02:00
public void processMonthlySettlements() {
    LocalDate lastMonth = LocalDate.now().minusMonths(1);

    List<Long> sellers = orderRepository
        .findDistinctSellersByMonth(lastMonth);

    for (Long sellerId : sellers) {
        SettlementPolicy policy = policyRepository.findBySellerId(sellerId);

        if (policy.getCycle() == SettlementCycle.MONTHLY) {
            Settlement settlement = calculateSettlement(sellerId, lastMonth);
            settlementRepository.save(settlement);

            // 정산 금액 송금
            transferService.transfer(settlement);

            // 세금계산서 발행
            taxInvoiceService.issue(settlement);
        }
    }
}
```

### 6.1 통합 검색 (Federated Search)

**현재 문제:** 상품, 경매, 사용자를 따로 검색

**표준:** 통합 검색 엔드포인트

```java
@RestController
public class UnifiedSearchController {

    @GetMapping("/api/search")
    public UnifiedSearchResult search(@RequestParam String q) {
        // 모든 도메인 병렬 검색
        CompletableFuture<List<Product>> products =
            CompletableFuture.supplyAsync(() -> productSearchService.search(q));

        CompletableFuture<List<Auction>> auctions =
            CompletableFuture.supplyAsync(() -> auctionSearchService.search(q));

        CompletableFuture<List<Member>> members =
            CompletableFuture.supplyAsync(() -> memberSearchService.search(q));

        CompletableFuture.allOf(products, auctions, members).join();

        return UnifiedSearchResult.builder()
            .products(products.get())
            .auctions(auctions.get())
            .members(members.get())
            .build();
    }
}
```

```java
public enum NotificationType {
    BID_OUTBID,        // 입찰 밀림
    AUCTION_WON,       // 낙찰
    ORDER_SHIPPED,     // 배송 시작
    PAYMENT_COMPLETED, // 결제 완료

@Service
public class NotificationService {

    public void send(Long memberId, NotificationType type, Map<String, Object> data) {
        Member member = memberRepository.findById(memberId);

        NotificationPreference pref = member.getNotificationPreference();

        // 사용자 설정에 따라 채널 선택
        if (pref.isPushEnabled()) {
            fcmService.sendPush(member.getDeviceToken(), buildPushMessage(type, data));
        }

        if (pref.isEmailEnabled() && type.isEmailWorthy()) {
            emailService.send(member.getEmail(), buildEmailTemplate(type, data));
        }

        if (pref.isSmsEnabled() && type.isUrgent()) {
            smsService.send(member.getPhoneNumber(), buildSmsMessage(type, data));
        }

        // 인앱 알림은 무조건 저장
        inAppNotificationRepository.save(
            InAppNotification.of(memberId, type, data)
        );
    }
}
```
